void setup()
{
  pinMode(A0, INPUT);
  pinMode(2, INPUT);
  pinMode(5, OUTPUT);
  pinMode(3, INPUT);
  pinMode(4, INPUT);
  pinMode(6, OUTPUT);
  pinMode(A1, INPUT);
  pinMode(7, OUTPUT);
}

void loop()
{
  if (analogRead(A0) <= 500 && digitalRead(2) == HIGH) {
    digitalWrite(5, HIGH);
  } else {
    digitalWrite(5, LOW);
  }
  if (digitalRead(3) == HIGH || digitalRead(4) == HIGH) {
    digitalWrite(6, HIGH);
  } else {
    digitalWrite(6, LOW);
  }
  if (analogRead(A1) > 250) {
    digitalWrite(7, HIGH);
  } else {
    digitalWrite(7, LOW);
  }
  delay(10); // Delay a little bit to improve simulation performance
